# Dépendances de l'agent — versions épinglées, conservatrices.
#
# Cibles vérifiées : Python 3.11 (Pi OS Bookworm) et 3.13 (Pi OS Trixie), 64 bits.
#
# `evdev` est une EXTENSION C sans roue précompilée pour aarch64 : pip la
# compile, ce qui exige python3-dev et un compilateur. Le paquet APT
# `python3-evdev` est déjà compilé pour la distribution — install.sh l'installe
# et crée le venv en --system-site-packages, puis ne demande à pip QUE les deux
# dépendances en Python pur ci-dessous. Ce fichier reste la référence complète
# (et le repli si APT ne fournit pas evdev).
#
# httpx et websockets publient des roues `py3-none-any` : aucune compilation,
# quelle que soit la version de Python (vérifié pour cp313/aarch64).
#
# Le reste de l'agent n'utilise que la bibliothèque standard (sqlite3, hmac,
# hashlib, zoneinfo, asyncio, http.server, socket) : les modules purs — et donc
# la totalité des tests — s'importent sans aucune de ces trois dépendances.

evdev==1.7.1
httpx==0.27.2
websockets==12.0
